def remove_dropout(model):
    for name, module in model.named_children():
        if isinstance(module, nn.Dropout):
            setattr(model, name, nn.Identity())
        else:
            remove_dropout(module)
    return model

header_tokenizer = PreTrainedTokenizerFast(tokenizer_file="header_tok.json")
header_tokenizer.add_special_tokens({'pad_token': '[PAD]'})
payload_tokenizer = PreTrainedTokenizerFast(tokenizer_file="payload_tok.json")
payload_tokenizer.add_special_tokens({'pad_token': '[PAD]'})

def create_string_embedding(header, payload, model,header_tokenizer, payload_tokenizer):
    header = header.lower()
    payload = payload.lower()
    inputs_header = header_tokenizer(header,return_tensors="pt",truncation=True, padding=True,max_length = 4096)
    inputs_payload = payload_tokenizer(payload,return_tensors="pt",truncation=True, padding=True,max_length = 4096)

    with torch.no_grad():

        outputs_header = model(**inputs_header)
        outputs_payload = model(**inputs_payload)

        cls_header = outputs_header.last_hidden_state[:,0,:]
        cls_payload = outputs_payload.last_hidden_state[:,0,:]

    return torch.cat([cls_header, cls_payload], dim=0).flatten().cpu().numpy()

async def make_prediction(name):
    REDIS_URL = os.getenv("REDIS_URL", "redis://0.0.0.0:6379")
    pool = redis.ConnectionPool.from_url(REDIS_URL)
    redis_client = redis.Redis(connection_pool=pool)
#    await redis_client.xgroup_create("suricata:packets", "redis-consumers")
    stream = "suricata:packets"


    try:
        data = await redis_client.xreadgroup("redis-consumers", name, streams = {stream:'>'},
        count=10,block=5000)

        for key, value in data:
            for i, j in value:
                 text = j.get(b'alert')
                 res = json.loads(text)
                 payload = res.pop("Payload")
                 header = res
                 emb = create_string_embedding(str(header), str(payload), model, header_tokenizer, payload_tokenizer)
                 copy = emb.tolist()
                 emb = emb.reshape(1,-1)
                 pred = random_forest.predict(emb)
                 return {
                     "src" : header.get("Source IP",  "N/A"),
                     "dst" : header.get("Destination IP",  "N/A"),
                     "emb" : copy,
                     "label" : int(pred),
                 }
#    return f"src: {src_ip}, dst: {dst_ip}, prediction: {pred}"
    except Exception as e:
            if "BUSYGROUP" in str(e):
                pass
            else:
                raise
dqueue = deque()
async def main(dq):
    consumers = ["c1", "c2", "c3"]
    while True:
          res = await asyncio.gather(*(make_prediction(name) for name in consumers))
          for r in res:
              dq.appendleft(r)
          await asyncio.sleep(1)

def runner(dq):
    asyncio.run(main(dq))

if "live_dq" not in st.session_state:
    st.session_state.live_dq = dqueue
if "thread_started" not in st.session_state:
    t = Thread(target=runner, args=(dqueue,), daemon=True)
    t.start()
    st.session_state.thread_started = True
